165template< UniformType T >
166void ShaderProgram::set_uniform(const std::string& name, const T& value) const
167{
168 const auto location{get_uniform_location(name)};
169
170 if constexpr (std::is_same_v< T, bool >) {
171 glUniform1i(location, value ? 1 : 0);
172 }
173 else if constexpr (std::is_same_v< T, int >) {
174 glUniform1i(location, value);
175 }
176 else if constexpr (std::is_same_v< T, float >) {
177 glUniform1f(location, value);
178 }
179 else if constexpr (std::is_same_v< T, glm::vec2 >) {
180 glUniform2fv(location, 1, glm::value_ptr(value));
181 }
182 else if constexpr (std::is_same_v< T, glm::vec3 >) {
183 glUniform3fv(location, 1, glm::value_ptr(value));
184 }
185 else if constexpr (std::is_same_v< T, glm::vec4 >) {
186 glUniform4fv(location, 1, glm::value_ptr(value));
187 }
188 else if constexpr (std::is_same_v< T, glm::mat2 >) {
189 glUniformMatrix2fv(location, 1, GL_FALSE, glm::value_ptr(value));
190 }
191 else if constexpr (std::is_same_v< T, glm::mat3 >) {
192 glUniformMatrix3fv(location, 1, GL_FALSE, glm::value_ptr(value));
193 }
194 else if constexpr (std::is_same_v< T, glm::mat4 >) {
195 glUniformMatrix4fv(location, 1, GL_FALSE, glm::value_ptr(value));
196 }
197 else if constexpr (IsTexture< T >::value) {
198 glUniform1i(location, value.get_last_texture_unit());
199 }
200 else {
201 static_assert(std::is_same_v< T, bool >, "Unsupported uniform type");
202 }
